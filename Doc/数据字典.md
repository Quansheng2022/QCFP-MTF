def generate_create_table(data_dict, table_name):
    """从数据字典生成建表语句"""
    columns = []
    for field in data_dict['table_schema']['columns']:
        col_def = f"  {field['name']} {field['type']}"
        if field.get('constraint'):
            col_def += f" {field['constraint']}"
        columns.append(col_def)
    
    return f"""
    CREATE TABLE {table_name} (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      {', '.join(columns)}
    );

def detect_schema_changes(old_dict, new_dict):
    """检测数据字典变化，生成迁移脚本"""
    old_fields = {f['name']: f for f in old_dict['table_schema']['columns']}
    new_fields = {f['name']: f for f in new_dict['table_schema']['columns']}
    
    changes = {
        'added': set(new_fields) - set(old_fields),
        'removed': set(old_fields) - set(new_fields),
        'modified': []
    }
    
    for name in set(old_fields) & set(new_fields):
        if old_fields[name]['type'] != new_fields[name]['type']:
            changes['modified'].append(name)
    
    return changes
