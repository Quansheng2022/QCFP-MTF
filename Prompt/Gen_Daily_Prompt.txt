# ==================== 辅助函数 ====================
def load_stock_data(ticker):
    """
    加载股票日线数据，返回 DataFrame（已排序）
    仅读取 OUTPUT_COLUMNS 中存在的列
    """
    file_path = os.path.join(DATA_DIR, f"{ticker}_daily_ta_mf_analysis.csv")
    try:
        sample = pd.read_csv(file_path, nrows=0)
        existing_cols = [c.replace(' ', '_') for c in sample.columns]
        cols_to_read = [col for col in OUTPUT_COLUMNS if col in existing_cols]
        if not cols_to_read:
            raise ValueError(f"文件 {file_path} 中无任何所需列")
        
        df = pd.read_csv(file_path, usecols=cols_to_read)
        df.columns = df.columns.str.replace(' ', '_')
        df['Date'] = pd.to_datetime(df['Date'])
        df = df.sort_values('Date').reset_index(drop=True)
        return df
    except Exception as e:
        print(f"读取 {ticker} 数据失败: {e}")
        return None

def get_recent_data(df, days):
    """获取最近 days 个交易日的数据切片，返回 (start_date, end_date, data_slice)"""
    if df is None or df.empty:
        return None, None, None
    data_slice = df.tail(days).copy()
    if data_slice.empty:
        return None, None, None
    start_date = data_slice['Date'].min()
    end_date = data_slice['Date'].max()
    return start_date, end_date, data_slice

def format_dataframe(df):
    """格式化数据表格：日期字符串、数值保留指定位数，空值处理为空字符串"""
    formatted = df.copy()
    formatted['Date'] = formatted['Date'].dt.strftime('%Y-%m-%d')
    
    format_rules = {
        'Close': '{:.2f}',
        'EMA5': '{:.3f}',
        'EMA20': '{:.3f}',
        'EMA100': '{:.3f}',
        'EMA200': '{:.3f}',
        'Volume_Ratio5': '{:.3f}',
        'MACD_Histogram': '{:.4f}',
        'Institutional_Flow': '{:,.0f}',
        'Individual_Flow': '{:,.0f}',
        'Inst_5MA': '{:,.0f}',
        'Ind_5MA': '{:,.0f}',
        'Turnover_Rate': '{:.2f}'
    }
    for col, fmt in format_rules.items():
        if col in formatted.columns:
            formatted[col] = formatted[col].apply(
                lambda x: fmt.format(x) if pd.notna(x) else ''
            )
    return formatted

def generate_question(stock_name, stock_code, start_date, end_date):
    """生成问题标题文本"""
    start_str = start_date.strftime('%Y年%m月%d日')
    end_str = end_date.strftime('%Y年%m月%d日')
    return f"问题: 对{stock_name} ({stock_code})近日走势进行复盘分析，复盘区间：{start_str} – {end_str}"

def write_and_print(output_file, text):
    """将文本同时写入文件和控制台"""
    print(text)
    output_file.write(text + "\n")

# ==================== 主流程 ====================
def main():
    # 确保输出目录存在
    output_dir = os.path.dirname(OUTPUT_FILEPATH)
    if output_dir and not os.path.exists(output_dir):
        os.makedirs(output_dir)
    
    # 打开输出文件（UTF-8 编码，确保中文正常）
    with open(OUTPUT_FILEPATH, 'w', encoding='utf-8') as f:
        # 写入标题
        header = "=" * 80 + "\n近日走势复盘分析 - 自动生成提问模板\n" + "=" * 80
        print(header)
        f.write(header + "\n\n")
        
        for stock in STOCK_LIST:
            code = stock["code"]
            name = stock["name"]
            log_msg = f"\n处理 {name} ({code}) ..."
            print(log_msg)
            f.write(log_msg + "\n")
            
            df = load_stock_data(code)
            if df is None:
                skip_msg = f"  跳过 {code}（数据加载失败）"
                print(skip_msg)
                f.write(skip_msg + "\n")
                continue
            
            start_date, end_date, data_slice = get_recent_data(df, REVIEW_DAYS)
            if data_slice is None or data_slice.empty:
                skip_msg = f"  跳过 {code}（无足够数据）"
                print(skip_msg)
                f.write(skip_msg + "\n")
                continue
            
            formatted_df = format_dataframe(data_slice)
            question = generate_question(name, code, start_date, end_date)
            
            # 输出问题及表格
            write_and_print(f, "\n" + question)
            # 将表格内容写入文件（使用 to_string 保持格式）
            table_str = formatted_df.to_string(index=False)
            write_and_print(f, table_str)
            write_and_print(f, "\n" + "-" * 80)
        
        # 写入完成提示
        completion_msg = f"\n输出已保存至：{OUTPUT_FILEPATH}"
        print(completion_msg)
        f.write(completion_msg + "\n")

if __name__ == "__main__":
    main()

def plot_daily_chart(ticker, df, output_pdf, start_date, end_date):
    """
    使用DataFrame数据绘制日线图表，包含价格、成交量、MACD和RSI指标
    """
    try:
        if df is None or df.empty:
            logger.error(f"没有数据可用于绘制 {ticker} 的图表")
            return

        logger.info(f"开始绘制 {ticker} 的日线图表...")
        
        # 1. 设置支持中文字符的字体
        plt.rcParams['font.sans-serif'] = ['Microsoft YaHei', 'SimHei', 'Arial Unicode MS']
        plt.rcParams['axes.unicode_minus'] = False
        plt.rcParams['font.weight'] = 'normal'

        # 2. 获取数据表中的实际日期范围 - 使用小写列名
        actual_start_date = df['date'].min()
        actual_end_date = df['date'].max()
        stock_name = df['stock_name'].iloc[0] if 'stock_name' in df.columns else ticker

        # 3. 创建PDF对象
        with PdfPages(output_pdf) as pdf:
            # ========== 第一页: 价格图表 + 指标图表 ==========
            # 4. 准备蜡烛图数据 - 使用小写列名
            df['date_num'] = df['date'].apply(mdates.date2num)
            ohlc = df[['date_num', 'open', 'high', 'low', 'close']].values

            # 5. 创建图表
            fig = plt.figure(figsize=(24, 24), dpi=100)
            gs = GridSpec(nrows=4, ncols=1, height_ratios=[4, 1.2, 1.2, 1.5], hspace=0.25)

            # 第一区域：价格图表
            ax1 = fig.add_subplot(gs[0])
            date_range_str = f"{actual_start_date.strftime('%Y-%m-%d')} 至 {actual_end_date.strftime('%Y-%m-%d')}"
            ax1.set_title(f'{ticker} {stock_name} 日线图 - {date_range_str}', fontsize=24)

            # 动态调整蜡烛图宽度
            num_points = len(df)
            base_width = 0.8
            candle_width = base_width * (0.5 if num_points>100 else 0.7 if num_points>50 else 0.9 if num_points>20 else 1)

            # 绘制蜡烛图
            candlestick_ohlc(ax1, ohlc, width=candle_width, colorup='g', colordown='r', alpha=0.8)

            # 绘制EMA均线 - 使用小写列名
            ema_cols = ['ema5', 'ema20', 'ema200']
            for ema in ema_cols:
                if ema in df.columns:
                    ax1.plot(df['date_num'], df[ema], label=ema.upper(), linewidth=2.5, alpha=0.9)
            ax1.set_ylabel('价格', fontsize=14)
            ax1.legend(loc='best', fontsize=12)
            ax1.grid(True, alpha=0.3)
            ax1.tick_params(axis='both', labelsize=12)
            ax1.set_xlim(df['date_num'].min()-candle_width*5, df['date_num'].max()+candle_width*5)

            # 第二区域：成交量
            ax2 = fig.add_subplot(gs[1], sharex=ax1)
            ax2.set_xlim(df['date_num'].min()-candle_width*5, df['date_num'].max()+candle_width*5)
            colors = ['g' if close>open_ else 'r' for open_, close in zip(df['open'], df['close'])]
            ax2.bar(df['date_num'], df['volume'], color=colors, width=candle_width, alpha=0.7, label='成交量')

            # 绘制成交量均线 - 使用小写列名
            if 'vol_ema5' in df.columns:
                ax2.plot(df['date_num'], df['vol_ema5'], color='blue', linewidth=2, label='5日成交量均线')
            if 'vol_ema20' in df.columns:
                ax2.plot(df['date_num'], df['vol_ema20'], color='gold', linewidth=2, label='20日成交量均线')
            ax2.set_ylabel('成交量', fontsize=14)
            ax2.grid(True, alpha=0.3)
            ax2.legend(loc='best', fontsize=12)

            # 第三区域：MACD - 使用小写列名
            ax3 = fig.add_subplot(gs[2], sharex=ax1)
            ax3.set_xlim(df['date_num'].min()-candle_width*5, df['date_num'].max()+candle_width*5)

            if 'macd_histogram' in df.columns:
                macd_colors = ['g' if val>0 else 'r' for val in df['macd_histogram']]
                ax3.bar(df['date_num'], df['macd_histogram'], color=macd_colors, width=candle_width, alpha=0.7, label='MACD')

            if 'macd_dif' in df.columns:
                ax3.plot(df['date_num'], df['macd_dif'], label='DIF', color='blue', linewidth=2.5, alpha=0.9)

            if 'macd_signal' in df.columns:
                ax3.plot(df['date_num'], df['macd_signal'], label='Signal', color='orange', linewidth=2.5, alpha=0.9)

            ax3.set_ylabel('MACD', fontsize=14)
            ax3.axhline(0, color='gray', linestyle='--', alpha=0.5)
            ax3.legend(loc='best', fontsize=12)
            ax3.grid(True, alpha=0.3)
            ax3.tick_params(axis='both', labelsize=12)

            # 第四区域：RSI - 使用小写列名
            ax4 = fig.add_subplot(gs[3], sharex=ax1)
            ax4.set_xlim(df['date_num'].min()-candle_width*5, df['date_num'].max()+candle_width*5)
            if 'rsi14' in df.columns:
                ax4.plot(df['date_num'], df['rsi14'], label='RSI14', color='blue', linewidth=2, alpha=0.9)
                ax4.axhline(70, color='red', linestyle='--', alpha=0.7, linewidth=1.5)
                ax4.axhline(30, color='green', linestyle='--', alpha=0.7, linewidth=1.5)
                ax4.axhline(50, color='gray', linestyle='-', alpha=0.5, linewidth=1.0)
                ax4.fill_between(df['date_num'], df['rsi14'], 70, where=df['rsi14']>=70, color='red', alpha=0.2)
                ax4.fill_between(df['date_num'], df['rsi14'], 30, where=df['rsi14']<=30, color='green', alpha=0.2)
                ax4.set_ylabel('RSI14', fontsize=14)
                ax4.set_ylim(0, 100)
                ax4.legend(loc='best', fontsize=12)
                ax4.grid(True, alpha=0.3)
                ax4.tick_params(axis='both', labelsize=12)
            else:
                ax4.axis('off')

            # 统一时间轴标签
            date_format = mdates.DateFormatter('%Y-%m-%d')
            tick_positions, tick_labels = [], []
            if len(df) < 30:
                tick_positions = df['date_num']
                tick_labels = [d.strftime('%Y-%m-%d') for d in df['date']]
            else:
                interval = max(1, len(df)//15)
                tick_positions = df['date_num'][::interval]
                tick_labels = [df['date'].iloc[i].strftime('%Y-%m-%d') for i in range(0, len(df), interval)]

            for ax in [ax1, ax2, ax3, ax4]:
                if ax.axison:
                    ax.xaxis.set_major_formatter(date_format)
                    ax.set_xticks(tick_positions)
                    plt.setp(ax.get_xticklabels(), rotation=30, ha='right', fontsize=10)

            plt.subplots_adjust(left=0.05, right=0.95, bottom=0.12, top=0.95, hspace=0.25)

            # 保存价格图表到PDF
            pdf_metadata = {
                'Title': f'{ticker} 日线分析图表',
                'Author': '量化分析系统',
                'Subject': f'{ticker} 技术分析 ({actual_start_date.strftime("%Y-%m-%d")} 至 {actual_end_date.strftime("%Y-%m-%d")})',
                'Keywords': f'股票分析,技术分析,EMA,MACD,RSI14,成交量,{ticker}',
                'CreationDate': pd.Timestamp.now()
            }
            pdf.savefig(fig, bbox_inches='tight', metadata=pdf_metadata)
            plt.close(fig)

            logger.info(f"图表已成功保存为PDF: {output_pdf}")

    except Exception as e:
        logger.error(f"处理数据或绘制图表时出错 - {ticker}: {str(e)}")
        traceback.print_exc()
        return


def display_indicator_summary(ticker, df, output_pdf, start_date, end_date):
    """
    使用DataFrame数据生成技术指标状态表格，结果追加到指定的PDF文件中
    """
    try:
        if df is None or df.empty:
            logger.error(f"没有数据可用于生成 {ticker} 的技术指标表格")
            return

        logger.info(f"开始生成 {ticker} 的技术指标表格...")

        # 1. 设置支持中文字符的字体
        plt.rcParams['font.sans-serif'] = ['SimHei']
        plt.rcParams['axes.unicode_minus'] = False

        # 2. 获取数据表中的实际日期范围 - 使用小写列名
        actual_start_date = df['date'].min()
        actual_end_date = df['date'].max()
        stock_name = df['stock_name'].iloc[0] if 'stock_name' in df.columns else ticker

        # 3. 创建技术指标表格图表
        fig_table = plt.figure(figsize=(24, 14), dpi=100)
        ax_table = fig_table.add_subplot(111)
        ax_table.axis('off')

        # 获取最后交易日数据
        last_row = df.iloc[-1]
        table_data = [["指标", "状态", "详细信息"]]
        streak_values = [None]

        # ===== 新增：处理布尔类型信号列 - 使用小写列名 =====
        bool_signal_columns = {
            'ema_golden_cross': 'EMA金叉信号',
            'ema_death_cross': 'EMA死叉信号',
            'macd_golden_cross': 'MACD金叉信号',
            'macd_death_cross': 'MACD死叉信号',
            'bottom_signal_cond': '底部信号',
            'top_signal_cond': '顶部信号',
            'bottom_entry_signal_cond': '底部入场信号',
            'top_exit_signal_cond': '顶部出场信号',
            'continuation_signal_cond': '趋势延续信号',
            'decline_continuation_cond': '下跌延续信号',
            'accumulation_cond': '底部吸筹信号',
            'distribution_cond': '顶部派发信号',
            'multi_timeframe_chip_confirm_bottom_cond': '多周期筹码底部确认',
            'multi_timeframe_chip_confirm_top_cond': '多周期筹码顶部确认'
        }
        
        # 检查并显示布尔信号列
        for col, label in bool_signal_columns.items():
            if col in df.columns:
                # 获取最新值（布尔类型）
                value = last_row.get(col, False)
                status = "TRUE" if value else "FALSE"
                
                # 如果是TRUE，计算持续天数
                if status == "TRUE":
                    # 查找第一次出现的位置
                    first_true_idx = None
                    for i in range(len(df)-1, -1, -1):
                        if df.iloc[i].get(col, False):
                            first_true_idx = i
                        else:
                            break
                    
                    if first_true_idx is not None:
                        first_date = df.iloc[first_true_idx]['date']
                        days = (df['date'].iloc[-1] - first_date).days
                        detail = f"{first_date.strftime('%Y-%m-%d')}起，持续{days}天"
                    else:
                        detail = "今日触发"
                else:
                    detail = ""
                
                table_data.append([label, status, detail])
                streak_values.append(None if status == "FALSE" else 1)

        # ===== EMA交叉指标 - 使用小写列名 =====
        ema_pairs = [("EMA5/10", "ema5_10_status", "ema5_10_streak"),
                     ("EMA5/20", "ema5_20_status", "ema5_20_streak"),
                     ("EMA10/20", "ema10_20_status", "ema10_20_streak"),
                     ("EMA20/50", "ema20_50_status", "ema20_50_streak"),
                     ("EMA20/200", "ema20_200_status", "ema20_200_streak"),
                     ("EMA50/200", "ema50_200_status", "ema50_200_streak")]
        for label, status_col, streak_col in ema_pairs:
            status = last_row.get(status_col, 'N/A')
            streak = last_row.get(streak_col, 'N/A')
            if status in ['Golden', 'Death']:
                table_data.append([label, status, f"持续{streak}天" if streak != 'N/A' else ''])
            streak_values.append(streak if status in ['Golden', 'Death'] else None)

        # ===== 均线突破确认 - 使用小写列名 =====
        ema20_break_above, start_date_above = check_breakthrough(df, 'ema20', 'above')
        if ema20_break_above:
            table_data.append(["确认突破EMA20", "TRUE", 
                              f"起始日：{start_date_above.strftime('%Y-%m-%d')}"])
            streak_values.append(None)

        ema20_break_below, start_date_below = check_breakthrough(df, 'ema20', 'below')
        if ema20_break_below:
            table_data.append(["确认跌破EMA20", "TRUE", 
                              f"起始日：{start_date_below.strftime('%Y-%m-%d')}"])
            streak_values.append(None)

        ema200_break_above, start_date_200_above = check_breakthrough(df, 'ema200', 'above')
        if ema200_break_above:
            table_data.append(["确认突破EMA200", "TRUE", 
                              f"起始日：{start_date_200_above.strftime('%Y-%m-%d')}"])
            streak_values.append(None)

        ema200_break_below, start_date_200_below = check_breakthrough(df, 'ema200', 'below')
        if ema200_break_below:
            table_data.append(["确认跌破EMA200", "TRUE", 
                              f"起始日：{start_date_200_below.strftime('%Y-%m-%d')}"])
            streak_values.append(None)

        # ===== EMA5/20/200黄金三角 - 使用小写列名 =====
        if len(df) > 0 and 'golden_triangle' in df.columns:
            # golden_triangle已经是布尔类型
            lookback_days = 30
            n = min(lookback_days, len(df))
            df_sub = df.tail(n).copy()
            golden_true = df_sub[df_sub['golden_triangle'] == True]
            if not golden_true.empty:
                last_date = df['date'].iloc[-1]
                first_golden_date = golden_true['date'].iloc[0]
                delta_days = (last_date - first_golden_date).days
                table_data.append([
                    "EMA5/20/200黄金三角", 
                    "TRUE", 
                    f"{first_golden_date.strftime('%Y-%m-%d')}，已出现{delta_days}天"
                ])
                streak_values.append(delta_days)

        # ===== EMA5/20/200死亡三角 - 使用小写列名 =====
        if len(df) > 0 and 'death_triangle' in df.columns:
            # death_triangle已经是布尔类型
            lookback_days = 30
            n = min(lookback_days, len(df))
            df_sub = df.tail(n).copy()
            death_true = df_sub[df_sub['death_triangle'] == True]
            if not death_true.empty:
                last_date = df['date'].iloc[-1]
                first_death_date = death_true['date'].iloc[0]
                delta_days = (last_date - first_death_date).days
                table_data.append([
                    "EMA5/20/200死亡三角", 
                    "TRUE", 
                    f"{first_death_date.strftime('%Y-%m-%d')}，已出现{delta_days}天"
                ])
                streak_values.append(delta_days)

        # ===== 均线趋势排列 - 使用小写列名 =====
        trend_ema = [
            ("短线多头", ['ema5', 'ema10', 'ema20'], lambda x: x[0]>x[1]>x[2]),
            ("短线空头", ['ema5', 'ema10', 'ema20'], lambda x: x[0]<x[1]<x[2]),
            ("长线多头", ['ema20', 'ema50', 'ema200'], lambda x: x[0]>x[1]>x[2]),
            ("长线空头", ['ema20', 'ema50', 'ema200'], lambda x: x[0]<x[1]<x[2])
        ]
        for label, ema_list, condition in trend_ema:
            ema_vals = [last_row.get(ema, float('nan')) for ema in ema_list]
            if not any(pd.isna(v) for v in ema_vals) and condition(ema_vals):
                table_data.append([label, "TRUE", ""])
                streak_values.append(None)

        # ===== EMA20上方阳线 - 使用小写列名 =====
        last_date = df['date'].iloc[-1]
        ema20_up_yang = None
        ema20_up_yang_count = 0

        for i in range(len(df)-1, len(df)-11, -1):
            if i < 1: continue
            row, prev_row = df.iloc[i], df.iloc[i-1]
            if row['close'] <= row['open']: continue

            if prev_row['close'] < prev_row['ema20'] and row['close'] > row['ema20']:
                ema20_up_yang = row['date'] if ema20_up_yang is None else ema20_up_yang
                ema20_up_yang_count += 1
            elif row['ema20'] <= row['open'] <= row['ema20']*1.03 and row['close'] >= row['open']:
                ema20_up_yang = row['date'] if ema20_up_yang is None else ema20_up_yang
                ema20_up_yang_count += 1

        if ema20_up_yang is not None:
            delta_days = (last_date - ema20_up_yang).days
            table_data.append([
                "EMA20上方阳线", 
                "TRUE", 
                f"{ema20_up_yang.strftime('%Y-%m-%d')}，已出现{delta_days}天，累计{ema20_up_yang_count}次"
            ])
            streak_values.append(None)

        # ===== EMA20下方阴线 - 使用小写列名 =====
        ema20_down_yin = None
        ema20_down_yin_count = 0

        for i in range(len(df)-1, len(df)-11, -1):
            if i < 1: continue
            row, prev_row = df.iloc[i], df.iloc[i-1]
            if row['close'] >= row['open']: continue

            if prev_row['close'] > prev_row['ema20'] and row['close'] < row['ema20']:
                ema20_down_yin = row['date'] if ema20_down_yin is None else ema20_down_yin
                ema20_down_yin_count += 1
            elif row['ema20']*0.97 <= row['open'] <= row['ema20'] and row['close'] <= row['open']:
                ema20_down_yin = row['date'] if ema20_down_yin is None else ema20_down_yin
                ema20_down_yin_count += 1

        if ema20_down_yin is not None:
            delta_days = (last_date - ema20_down_yin).days
            table_data.append([
                "EMA20下方阴线", 
                "TRUE", 
                f"{ema20_down_yin.strftime('%Y-%m-%d')}，已出现{delta_days}天，累计{ema20_down_yin_count}次"
            ])
            streak_values.append(None)

        # ===== 跳空高开阳线 - 使用小写列名 =====
        up_red_dates = []
        start_idx = max(0, len(df)-10)
        for i in range(start_idx, len(df)):
            if i < 1: 
                continue
            row, prev_row = df.iloc[i], df.iloc[i-1]
            if row['open'] > prev_row['high'] and row['close'] > row['open']:
                up_red_dates.append(row['date'])

        if up_red_dates:
            up_red_dates.sort()
            last_occurrence = up_red_dates[-1].strftime('%Y-%m-%d')
            table_data.append([
                "跳空高开阳线", 
                "TRUE", 
                f"10日数量: {len(up_red_dates)}，最近出现: {last_occurrence}"
            ])
            streak_values.append(None)

        # ===== 跳空低开阴线 - 使用小写列名 =====
        down_black_dates = []
        for i in range(start_idx, len(df)):
            if i < 1: 
                continue
            row, prev_row = df.iloc[i], df.iloc[i-1]
            if row['open'] < prev_row['low'] and row['close'] < row['open']:
                down_black_dates.append(row['date'])

        if down_black_dates:
            down_black_dates.sort()
            last_occurrence = down_black_dates[-1].strftime('%Y-%m-%d')
            table_data.append([
                "跳空低开阴线", 
                "TRUE", 
                f"10日数量: {len(down_black_dates)}，最近出现: {last_occurrence}"
            ])
            streak_values.append(None)

        # ===== 多均线粘合 - 使用小写列名 =====
        ema_columns = ['ema5', 'ema10', 'ema20', 'ema50', 'ema60', 'ema100', 'ema200']
        if all(col in df.columns for col in ema_columns):
            recent_data = df.tail(10).copy()
            convergence_dates = []
            convergence_groups = []

            for idx, row in recent_data.iterrows():
                date = row['date']
                ema_values = {col: row[col] for col in ema_columns if not pd.isna(row[col])}
                convergence_groups_for_day = find_convergence_groups_clustering(ema_values, threshold=0.005)
                significant_groups = [group for group in convergence_groups_for_day if len(group) >= 3]
                if significant_groups:
                    convergence_dates.append(date)
                    convergence_groups.append({
                        'date': date,
                        'groups': significant_groups,
                        'largest_group': max(significant_groups, key=len) if significant_groups else []
                    })

            if convergence_dates:
                convergence_dates.sort()
                last_convergence_date = convergence_dates[-1].strftime('%Y-%m-%d')
                last_detail = convergence_groups[-1]
                largest_group = last_detail['largest_group']
                sorted_group = sort_ema_by_period(largest_group)
                group_str = "/".join(sorted_group)
                detail_str = f"最近日期: {last_convergence_date}，粘合均线组: {group_str}（{len(largest_group)}条）"
                table_data.append(["多均线粘合", "TRUE", detail_str])
                streak_values.append(None)

        # ===== 一阳穿越多条均线 - 使用小写列名 =====
        if all(col in df.columns for col in ema_columns):
            recent_data = df.tail(10).copy()
            cross_dates = []
            cross_details = []

            for idx, row in recent_data.iterrows():
                date = row['date']
                if row['close'] <= row['open']:
                    continue
                open_price = row['open']
                close_price = row['close']
                cross_count = 0
                crossed_emas = []

                for ema_col in ema_columns:
                    ema_value = row[ema_col]
                    if pd.isna(ema_value):
                        continue
                    if open_price <= ema_value <= close_price:
                        cross_count += 1
                        crossed_emas.append(ema_col.upper())

                if cross_count >= 2:
                    cross_dates.append(date)
                    cross_details.append({
                        'date': date,
                        'emas': crossed_emas,
                        'count': cross_count
                    })

            if cross_dates:
                cross_dates.sort()
                last_cross_date = cross_dates[-1].strftime('%Y-%m-%d')
                last_detail = cross_details[-1]
                ema_names = ", ".join(last_detail['emas'][:3])
                if len(last_detail['emas']) > 3:
                    ema_names += f"等{last_detail['count']}条均线"
                else:
                    ema_names = f"{last_detail['count']}条均线: {ema_names}"
                detail_str = f"最近日期: {last_cross_date}，穿越{ema_names}"
                table_data.append(["一阳穿越多条均线", "TRUE", detail_str])
                streak_values.append(None)

        # ===== 一阴穿越多条均线 - 使用小写列名 =====
        if all(col in df.columns for col in ema_columns):
            recent_data = df.tail(10).copy()
            cross_dates = []
            cross_details = []

            for idx, row in recent_data.iterrows():
                date = row['date']
                if row['close'] >= row['open']:
                    continue
                open_price = row['open']
                close_price = row['close']
                cross_count = 0
                crossed_emas = []

                for ema_col in ema_columns:
                    ema_value = row[ema_col]
                    if pd.isna(ema_value):
                        continue
                    if close_price <= ema_value <= open_price:
                        cross_count += 1
                        crossed_emas.append(ema_col.upper())

                if cross_count >= 2:
                    cross_dates.append(date)
                    cross_details.append({
                        'date': date,
                        'emas': crossed_emas,
                        'count': cross_count
                    })

            if cross_dates:
                cross_dates.sort()
                last_cross_date = cross_dates[-1].strftime('%Y-%m-%d')
                last_detail = cross_details[-1]
                ema_names = ", ".join(last_detail['emas'][:3])
                if len(last_detail['emas']) > 3:
                    ema_names += f"等{last_detail['count']}条均线"
                else:
                    ema_names = f"{last_detail['count']}条均线: {ema_names}"
                detail_str = f"最近日期: {last_cross_date}，穿越{ema_names}"
                table_data.append(["一阴穿越多条均线", "TRUE", detail_str])
                streak_values.append(None)

        # ===== KDJ线金叉/死叉 - 使用小写列名 =====
        has_kdj = all(col in df.columns for col in ['kdj_k', 'kdj_d', 'kdj_j'])
        kdj_golden, kdj_death = None, None

        if has_kdj:
            lookback_window = df.tail(10).copy()
            for i in range(1, len(lookback_window)):
                prev, curr = lookback_window.iloc[i-1], lookback_window.iloc[i]
                if curr['kdj_j'] <= 30 and prev['kdj_k'] < prev['kdj_d'] and curr['kdj_k'] > curr['kdj_d']:
                    kdj_golden = {'date': curr['date'], 'days': (df['date'].iloc[-1] - curr['date']).days}
                    break
                if curr['kdj_j'] >= 70 and prev['kdj_k'] > prev['kdj_d'] and curr['kdj_k'] < curr['kdj_d']:
                    kdj_death = {'date': curr['date'], 'days': (df['date'].iloc[-1] - curr['date']).days}
                    break

        if kdj_golden:
            table_data.append([
                "KDJ线金叉", "TRUE", 
                f"金叉日期: {kdj_golden['date'].strftime('%Y-%m-%d')}, 持续天数: {kdj_golden['days']}"
            ])
            streak_values.append(kdj_golden['days'])

        if kdj_death:
            table_data.append([
                "KDJ线死叉", "TRUE", 
                f"死叉日期: {kdj_death['date'].strftime('%Y-%m-%d')}, 持续天数: {kdj_death['days']}"
            ])
            streak_values.append(kdj_death['days'])

        # ===== MACD - 使用小写列名 =====
        macd_status = last_row.get('macd_status', 'N/A')
        macd_value = f"{last_row.get('macd', 'N/A'):.4f}" if 'macd' in df.columns else 'N/A'
        macd_details = macd_value
        if macd_status == 'Golden':
            macd_details += f"/金叉持续: {last_row.get('macd_golden_streak', 0)}天"
        elif macd_status == 'Death':
            macd_details += f"/死叉持续: {last_row.get('macd_death_streak', 0)}天"
        table_data.append(["MACD", macd_status, macd_details])
        streak_values.append(last_row.get('macd_golden_streak', 0) if macd_status=='Golden' else 
                            last_row.get('macd_death_streak', 0) if macd_status=='Death' else None)

        # ===== 成交量 - 使用小写列名 =====
        volume = f"{last_row['volume']:,.0f}"
        volume_ratio5 = f"{last_row.get('volume_ratio5', 1):.2f}x" if 'volume_ratio5' in df.columns else 'N/A'
        table_data.append(["成交量", volume, volume_ratio5])
        streak_values.append(None)

        # ===== 成交量金叉/死叉 - 使用小写列名 =====
        if all(col in df.columns for col in ['vol_ema5', 'vol_ema20']):
            recent_data = df.tail(10).copy()
            golden_cross_dates = []
            death_cross_dates = []

            for i in range(1, len(recent_data)):
                prev_row = recent_data.iloc[i-1]
                curr_row = recent_data.iloc[i]
                if prev_row['vol_ema5'] <= prev_row['vol_ema20'] and curr_row['vol_ema5'] > curr_row['vol_ema20']:
                    golden_cross_dates.append(curr_row['date'])
                if prev_row['vol_ema5'] >= prev_row['vol_ema20'] and curr_row['vol_ema5'] < curr_row['vol_ema20']:
                    death_cross_dates.append(curr_row['date'])

            if golden_cross_dates:
                golden_cross_dates.sort()
                last_golden_date = golden_cross_dates[-1].strftime('%Y-%m-%d')
                count = len(golden_cross_dates)
                table_data.append([
                    "成交量金叉", 
                    "TRUE", 
                    f"10日内发生{count}次，最近日期: {last_golden_date}"
                ])
                streak_values.append(None)

            if death_cross_dates:
                death_cross_dates.sort()

def find_convergence_groups_clustering(ema_values, threshold=0.005):
    """
    使用聚类算法检测均线粘合
    """
    if not ema_values:
        return []

    sorted_emas = sorted(ema_values.items(), key=lambda x: x[1])
    groups = []
    current_group = [sorted_emas[0][0]]

    for i in range(1, len(sorted_emas)):
        current_ema_name, current_ema_value = sorted_emas[i]
        prev_ema_name, prev_ema_value = sorted_emas[i-1]
        max_val = max(current_ema_value, prev_ema_value)
        min_val = min(current_ema_value, prev_ema_value)
        if max_val == 0:
            continue
        diff_ratio = (max_val - min_val) / max_val

        if diff_ratio < threshold:
            if current_ema_name not in current_group:
                current_group.append(current_ema_name)
            if prev_ema_name not in current_group:
                current_group.append(prev_ema_name)
        else:
            if len(current_group) >= 2:
                groups.append(current_group)
            current_group = [current_ema_name]

    if len(current_group) >= 2:
        groups.append(current_group)

    return merge_overlapping_groups(groups)


def merge_overlapping_groups(groups):
    """
    合并有重叠的均线组
    """
    if not groups:
        return []

    merged = True
    current_groups = groups.copy()

    while merged:
        merged = False
        new_groups = []
        used = [False] * len(current_groups)

        for i in range(len(current_groups)):
            if used[i]:
                continue
            group1 = set(current_groups[i])
            merged_group = set(group1)
            used[i] = True

            for j in range(i+1, len(current_groups)):
                if used[j]:
                    continue
                group2 = set(current_groups[j])
                if group1 & group2:
                    merged_group |= group2
                    used[j] = True
                    merged = True

            new_groups.append(list(merged_group))

        for i in range(len(current_groups)):
            if not used[i]:
                new_groups.append(current_groups[i])

        current_groups = new_groups

    return current_groups


def sort_ema_by_period(ema_list):
    """
    按周期顺序对均线进行排序
    """
    period_order = {
        'ema5': 0,
        'ema10': 1,
        'ema20': 2,
        'ema50': 3,
        'ema60': 4,
        'ema100': 5,
        'ema200': 6
    }
    return sorted(ema_list, key=lambda x: period_order.get(x, 999))


def check_breakthrough(df, ema_col, direction='above'):
    """
    检查是否确认突破/跌破指定EMA均线
    """
    if len(df) < 10:
        return False, None

    recent_data = df.tail(10).copy()

    if direction == 'above':
        above_count = 0
        start_date = None
        for i in range(len(recent_data)-1, -1, -1):
            if recent_data.iloc[i]['close'] > recent_data.iloc[i][ema_col]:
                above_count += 1
                if above_count == 1:
                    start_date = recent_data.iloc[i]['date']
                if above_count >= 3:
                    if i > 0 and recent_data.iloc[i-1]['close'] <= recent_data.iloc[i-1][ema_col]:
                        return True, start_date
            else:
                above_count = 0
                start_date = None
    else:
        below_count = 0
        start_date = None
        for i in range(len(recent_data)-1, -1, -1):
            if recent_data.iloc[i]['close'] < recent_data.iloc[i][ema_col]:
                below_count += 1
                if below_count == 1:
                    start_date = recent_data.iloc[i]['date']
                if below_count >= 3:
                    if i > 0 and recent_data.iloc[i-1]['close'] >= recent_data.iloc[i-1][ema_col]:
                        return True, start_date
            else:
                below_count = 0
                start_date = None

    return False, None


def merge_pdfs(pdf_inputs, pdf_output):
    """
    将多个PDF文件合并为一个PDF文件
    """
    merger = PyPDF2.PdfMerger()
    merged_count = 0
    missing_files = []

    try:
        for pdf in pdf_inputs:
            if not os.path.exists(pdf):
                logger.warning(f"{pdf} 文件不存在，跳过合并")
                missing_files.append(pdf)
                continue

            try:
                with open(pdf, 'rb') as f:
                    merger.append(f)
                merged_count += 1
            except Exception as e:
                logger.error(f"处理文件 {pdf} 时出错: {str(e)}")
                raise

        if merged_count > 0:
            with open(pdf_output, 'wb') as output_file:
                merger.write(output_file)
            logger.info(f"成功合并 {merged_count} 个PDF文件到: {pdf_output}")
        else:
            logger.warning("没有有效的PDF文件可供合并")

    except Exception as e:
        logger.error(f"合并PDF时出错: {str(e)}")
        raise
    finally:
        merger.close()

    return merged_count, missing_files
